\begin{frame}{索引：隐藏约束 vs. 输出压缩}
\scriptsize
\textbf{本页是机制对照，不重复展示例子。}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\linewidth}
\textbf{隐藏约束路线}
\begin{itemize}
  \item 看 Case B：B5 hidden refusal、B3 hidden policy。
  \item 完整四件套：prompt 要求不明说拒绝/政策，回复保持安全教育或 broad guidance，token 位于 \code{event\_window} / \code{reply\_early}，decode 读到 refusal / policy / redirect。
  \item 解释边界：B2/B4 是弱信号负例，不能泛化成所有 hidden motivation 都可读。
\end{itemize}
\end{column}
\begin{column}{0.48\linewidth}
\textbf{输出压缩路线}
\begin{itemize}
  \item 看 Case D：D4 final-only；Case E：R1 5-word compression。
  \item 完整四件套：prompt 强制短输出，回复极短，token 集中在少数 reply positions，decode 漂移或降级。
  \item 解释边界：压缩失败是 NLA 可靠性问题，不等价于模型内部没有相关知识。
\end{itemize}
\end{column}
\end{columns}

\vspace{0.35em}
\alert{并排结论：}隐藏约束场景仍可能有可读信号；极端输出压缩会直接降低 NLA round trip 的可靠性。
\end{frame}


\begin{frame}{索引：语言切换 vs. 评测感知}
\scriptsize
\textbf{本页是机制对照，不重复展示例子。}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\linewidth}
\textbf{语言路线}
\begin{itemize}
  \item 看 Case C：C3 SWITCH、C4 implicit language、C5 English-Q Chinese-A。
  \item 完整四件套：prompt 指定语言或文化 cue，回复呈现对应语言，token 覆盖 \code{event\_window} / \code{reply\_early}，decode 读到 Chinese / English / translation。
  \item 解释边界：C4 证明 language hit 可能来自文化语义邻域。
\end{itemize}
\end{column}
\begin{column}{0.48\linewidth}
\textbf{评测感知路线}
\begin{itemize}
  \item 看 Case A：A3 safety evaluation；Case F：F1 sampling。
  \item 完整四件套：prompt 明示 evaluation，回复给安全评估式回答，token 位置随 strategy 改变，decode 读到 evaluation / policy / safety。
  \item 解释边界：A2 说明低 cos 不等于 awareness，F1 说明 prompt-final 未必命中。
\end{itemize}
\end{column}
\end{columns}

\vspace{0.35em}
\alert{并排结论：}语言信号强，但容易被文化 cue 混淆；评测感知信号弱，必须报告采样窗口。
\end{frame}
